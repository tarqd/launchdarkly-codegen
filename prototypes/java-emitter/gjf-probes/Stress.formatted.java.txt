import com.launchdarkly.sdk.server.interfaces.LDClientInterface;

/** Stress. */
public final class Stress {
  /** The wrapped SDK client. */
  private final LDClientInterface client;

  /**
   * Wraps.
   *
   * @param client c
   */
  public Stress(LDClientInterface client) {
    this.client = client;
  }

  /**
   * Checkout V2
   *
   * <p>This is a very long description that keeps going well past the one hundred column limit so
   * that we can see exactly how the formatter chooses to wrap words across lines in a paragraph. A
   * second source line that should join the first.

   * <p>Ünïcödé wörds ünïcödé wörds ünïcödé wörds ünïcödé wörds ünïcödé wörds ünïcödé wörds ünïcödé.

   * https://app.eu.launchdarkly.com/projects/storefront-with-a-long-name/flags/checkout-v2-with-an-extremely-long-key-name
   *
   * @param context the context to evaluate the flag for
   * @param fallback the value to return if the flag can't be evaluated

   *     stop compiling.

      LDContext context, LDValue fallback) {

    return client.jsonValueVariationDetail(
        Keys.CHECKOUT_V2_WITH_AN_EXTREMELY_LONG_KEY_NAME_AND_SOME_MORE_TEXT_HERE,
        context,
        fallback);
  }

  /** Keys. */
  public static final class Keys {
    private Keys() {}

    /**
     * {@code checkout-v2-with-an-extremely-long-key-name-and-yet-more-text-to-pass-a-hundred-cols}
     */
